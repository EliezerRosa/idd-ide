// Puramente determinístico, teste de intersecção temporal
export const verificarOverlap = (
  agenda: ReadonlyArray<Slot>,
  novoSlot: Readonly<Slot>
): boolean => {
  return agenda.some(s => max(s.inicio, novoSlot.inicio) < min(s.fim, novoSlot.fim));
};
