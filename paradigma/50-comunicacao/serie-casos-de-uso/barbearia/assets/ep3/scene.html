-- INVARIANTE SAGRADA 1: CADEIRA ÚNICA CREATE OR REPLACE FUNCTION fn_check_cadeira_unica() RETURNS TRIGGER AS $$ BEGIN -- Verifica se já existe agendamento ativo no mesmo horário IF EXISTS ( SELECT 1 FROM agendamentos WHERE profissional_id = NEW.profissional_id AND status != 'CANCELADO' AND tsrange(data_hora_inicio, data_hora_fim) && tsrange(NEW.data_hora_inicio, NEW.data_hora_fim) ) THEN RAISE EXCEPTION 'VIOLACAO_INVARIANTE_CADEIRA_UNICA: Barbeiro ocupado!'; END IF; RETURN NEW; END; $$ LANGUAGE plpgsql;

export type Status = 'RESERVADO' | 'CONFIRMADO' | 'NA_CADEIRA' | 'CONCLUIDO'; export class AgendamentoDominio { public status: Status = 'RESERVADO'; iniciarAtendimento(): void { if (this.status !== 'CONFIRMADO') { throw new Error('Apenas reservas confirmadas vão para a cadeira!'); } this.status = 'NA_CADEIRA'; } concluirAtendimento(): void { if (this.status !== 'NA_CADEIRA') { throw new Error('Não é possível concluir sem passar pela cadeira!'); } this.status = 'CONCLUIDO'; } }
Cálculo de Comissão em Centavos Precisão Monetária
// Cálculo matemático puro em centavos inteiros export function calcularComissaoCentavos( precoTotalCentavos: number, comissaoPct: number ): { barbeiroCentavos: number; casaCentavos: number } { if (comissaoPct < 0 || comissaoPct > 100) { throw new Error('Percentual de comissão inválido'); } // Math.floor garante conservação monetária sem dízima const barbeiroCentavos = Math.floor((precoTotalCentavos * comissaoPct) / 100); const casaCentavos = precoTotalCentavos - barbeiroCentavos; return { barbeiroCentavos, casaCentavos }; }
