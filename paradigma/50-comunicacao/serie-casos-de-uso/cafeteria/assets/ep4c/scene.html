class ServicoCozinhaCafeteria {
  @invariant({
    pre: (p, lotes) => p.itens.every(i => hasSaldoSuficiente(i.fichaTecnica, lotes)),
    post: (resultado) => resultado.custoInsumos === somarCustosFichaTecnica(resultado),
    onError: () => new InsufficientStockException("Estoque insuficiente de grãos ou insumos!")
  })
  public static processarPedidoBalcao(
    pedido: PedidoCafeteria,
    lotesEstoque: ReadonlyArray<LoteEstoque>
  ): ComandaLiquidada {
    return pedido.consumirReceitaEfetiva(lotesEstoque);
  }
}
