// Transição rigorosa de status da Comanda
public transicionarPara(novoStatus: StatusComanda): void {
  if (!transicoesValidas[this.status].includes(novoStatus)) {
    throw new Error("Transição proibida");
  }

  // INVARIANTE SAGRADA DE LIQUIDAÇÃO PRÉVIA
  if (novoStatus === 'ENTREGUE' && !this.pago) {
    throw new Error("Impossível entregar comanda não paga!");
  }
  this.status = novoStatus;
}
