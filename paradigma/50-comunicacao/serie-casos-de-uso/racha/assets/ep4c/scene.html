function dividirConta(mesa) {
  let total = mesa.valorTotal; // float impreciso
  let cadaUm = total / mesa.amigos.length;
  mesa.amigos.forEach(a => {
    a.deve = cadaUm; // Perda de centavos em dízimas!
  });
  return mesa;
